# ClipRTC Core v2 Phase 0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create the `cliprtc-core` crate, deterministic room CA, per-device identity, trust store, protocol envelope/codec, Lamport clock, dedup window, and PR CI — with no Tauri dependency.

**Architecture:** Add a Cargo workspace under `src-tauri/`. `cliprtc-core` owns identity/protocol primitives and can be tested headlessly. The Tauri crate remains the shell until later phases.

**Tech Stack:** Rust 2021, `rcgen`, `rustls-pki-types`, `x509-parser`, `ed25519-dalek`, `hkdf`, `sha2`, `serde`, `uuid`, `tokio`, `thiserror`.

## Global Constraints

- LAN-only desktop app; no NAT traversal, mobile, accounts, or cloud.
- Room key is shared; default generated key must be high entropy.
- Device private key never leaves the machine; tests may use in-memory keys.
- Core must not depend on `tauri`, window, tray, or WebView.
- No runtime `unwrap()` / `expect()` in library code.
- No global `Lazy` / `OnceCell` business state.
- Every network/file operation gets a timeout and cancellation in later phases.
- Keep Conventional Commits (`feat:`, `fix:`, `chore(ci):`).

```rust
use hkdf::Hkdf;
use rcgen::{
    BasicConstraints, CertificateParams, CertifiedIssuer, DnType, IsCa, KeyPair,
    KeyUsagePurpose, PKCS_ED25519,
};
use rustls_pki_types::{CertificateDer, PrivatePkcs8KeyDer};
use sha2::{Digest, Sha256};
use ed25519_dalek::pkcs8::EncodePrivateKey;

use crate::error::{CoreError, Result};

pub struct RoomCa {
    issuer: CertifiedIssuer<'static, KeyPair>,
}

impl RoomCa {
    pub fn derive(room_key: &str) -> Result<Self> {
        let mut seed = [0u8; 32];
        Hkdf::<Sha256>::new(Some(b"cliprtc-room-ca-v1"), room_key.as_bytes())
            .expand(b"ed25519-seed", &mut seed)
            .map_err(|_| CoreError::Identity("hkdf expand failed".into()))?;

        let signing = ed25519_dalek::SigningKey::from_bytes(&seed);
        let pkcs8 = signing
            .to_pkcs8_der()
            .map_err(|e| CoreError::Identity(format!("pkcs8 encode failed: {e}")))?;
        let key = KeyPair::from_der_and_sign_algo(
            &PrivatePkcs8KeyDer::from(pkcs8.as_bytes().to_vec()),
            &PKCS_ED25519,
        )
        .map_err(|e| CoreError::Identity(format!("ca key load failed: {e}")))?;

        let mut params = CertificateParams::new(Vec::<String>::new())
            .map_err(|e| CoreError::Identity(format!("ca params failed: {e}")))?;
        params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
        params.key_usages = vec![
            KeyUsagePurpose::KeyCertSign,
            KeyUsagePurpose::DigitalSignature,
        ];
        params
            .distinguished_name
            .push(DnType::CommonName, "ClipRTC Room CA");

        let issuer = CertifiedIssuer::self_signed(params, key)
            .map_err(|e| CoreError::Identity(format!("ca sign failed: {e}")))?;
        Ok(Self { issuer })
    }

    pub fn cert_der(&self) -> &CertificateDer<'static> {
        self.issuer.as_ref().der()
    }

    pub fn fingerprint(&self) -> [u8; 32] {
        Sha256::digest(self.cert_der().as_ref()).into()
    }
}
```

```rust
use rcgen::string::Ia5String;
use rcgen::{
    CertificateParams, DnType, KeyPair, KeyUsagePurpose, SanType, PKCS_ED25519,
};
use rustls_pki_types::{CertificateDer, PrivatePkcs8KeyDer};
use sha2::{Digest, Sha256};
use uuid::Uuid;
use x509_parser::prelude::*;

use crate::error::{CoreError, Result};
use crate::identity::ca::RoomCa;

pub struct DeviceIdentity {
    pub uuid: Uuid,
    key: KeyPair,
}

pub struct DeviceCert {
    pub cert_der: CertificateDer<'static>,
    pub key_der: PrivatePkcs8KeyDer<'static>,
}

impl DeviceIdentity {
    pub fn generate(uuid: Uuid) -> Result<Self> {
        let key = KeyPair::generate_for(&PKCS_ED25519)
            .map_err(|e| CoreError::Identity(format!("device key failed: {e}")))?;
        Ok(Self { uuid, key })
    }

    pub fn from_pkcs8(uuid: Uuid, der: &[u8]) -> Result<Self> {
        let key = KeyPair::from_der_and_sign_algo(
            &PrivatePkcs8KeyDer::from(der.to_vec()),
            &PKCS_ED25519,
        )
        .map_err(|e| CoreError::Identity(format!("device key load failed: {e}")))?;
        Ok(Self { uuid, key })
    }

    pub fn private_key_der(&self) -> Vec<u8> {
        self.key.serialize_der()
    }

    pub fn public_key_hash(&self) -> [u8; 32] {
        Sha256::digest(self.key.public_key_raw()).into()
    }
}

impl RoomCa {
    pub fn issue_device_cert(&self, device: &DeviceIdentity) -> Result<DeviceCert> {
        let mut params = CertificateParams::new(vec!["cliprtc.local".to_string()])
            .map_err(|e| CoreError::Identity(format!("device params failed: {e}")))?;
        let uri = Ia5String::try_from(format!("cliprtc://device/{}", device.uuid))
            .map_err(|e| CoreError::Identity(format!("device uri failed: {e}")))?;
        params.subject_alt_names.push(SanType::URI(uri));
        params.key_usages = vec![KeyUsagePurpose::DigitalSignature];
        params
            .distinguished_name
            .push(DnType::CommonName, device.uuid.to_string());

        let cert = params
            .signed_by(&device.key, &self.issuer)
            .map_err(|e| CoreError::Identity(format!("device cert failed: {e}")))?;
        Ok(DeviceCert {
            cert_der: cert.der().clone(),
            key_der: PrivatePkcs8KeyDer::from(device.key.serialize_der()),
        })
    }
}

pub fn extract_device_uuid(cert_der: &[u8]) -> Result<Uuid> {
    let (_, cert) = X509Certificate::from_der(cert_der)
        .map_err(|e| CoreError::Identity(format!("parse cert failed: {e}")))?;

    if let Ok(Some(san)) = cert.subject_alternative_name() {
        for name in &san.value.general_names {
            if let GeneralName::URI(uri) = name {
                if let Some(id) = uri.strip_prefix("cliprtc://device/") {
                    return Uuid::parse_str(id)
                        .map_err(|e| CoreError::Identity(format!("bad uuid: {e}")));
                }
            }
        }
    }

    for attr in cert.subject().iter_common_name() {
        if let Ok(id) = attr.as_str() {
            if let Ok(uuid) = Uuid::parse_str(id) {
                return Ok(uuid);
            }
        }
    }

    Err(CoreError::Identity("certificate has no device uuid".into()))
}

pub fn extract_public_key_hash(cert_der: &[u8]) -> Result<[u8; 32]> {
    let (_, cert) = X509Certificate::from_der(cert_der)
        .map_err(|e| CoreError::Identity(format!("parse cert failed: {e}")))?;
    let raw = cert.public_key().subject_public_key.data.as_ref();
    Ok(Sha256::digest(raw).into())
}
```

```rust
use std::collections::HashMap;
use std::time::SystemTime;

use uuid::Uuid;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PeerTrust {
    pub public_key_hash: [u8; 32],
    pub first_seen: SystemTime,
    pub blocked: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TrustDecision {
    New,
    Known,
    Changed,
    Blocked,
}

#[derive(Debug, Default)]
pub struct TrustStore {
    peers: HashMap<Uuid, PeerTrust>,
}

impl TrustStore {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn check_and_record(&mut self, uuid: Uuid, public_key_hash: [u8; 32]) -> TrustDecision {
        if let Some(existing) = self.peers.get_mut(&uuid) {
            if existing.blocked {
                return TrustDecision::Blocked;
            }
            if existing.public_key_hash == public_key_hash {
                return TrustDecision::Known;
            }
            return TrustDecision::Changed;
        }

        self.peers.insert(
            uuid,
            PeerTrust {
                public_key_hash,
                first_seen: SystemTime::now(),
                blocked: false,
            },
        );
        TrustDecision::New
    }

    pub fn block(&mut self, uuid: Uuid) {
        if let Some(peer) = self.peers.get_mut(&uuid) {
            peer.blocked = true;
        }
    }

    pub fn get(&self, uuid: &Uuid) -> Option<&PeerTrust> {
        self.peers.get(uuid)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn first_seen_is_new_then_known() {
        let mut store = TrustStore::new();
        let id = Uuid::new_v4();
        assert_eq!(store.check_and_record(id, [1; 32]), TrustDecision::New);
        assert_eq!(store.check_and_record(id, [1; 32]), TrustDecision::Known);
    }

    #[test]
    fn changed_key_is_rejected() {
        let mut store = TrustStore::new();
        let id = Uuid::new_v4();
        store.check_and_record(id, [1; 32]);
        assert_eq!(store.check_and_record(id, [2; 32]), TrustDecision::Changed);

    fn blocked_peer_is_rejected() {
        let mut store = TrustStore::new();
        let id = Uuid::new_v4();
        store.check_and_record(id, [1; 32]);
        store.block(id);
        assert_eq!(store.check_and_record(id, [1; 32]), TrustDecision::Blocked);
```

```rust
use serde::{Deserialize, Serialize};
use uuid::Uuid;

use crate::error::{CoreError, Result};

pub const PROTOCOL_VERSION: u16 = 2;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum EventKind {
    Text,
    Image,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct ClipboardEvent {
    pub protocol_version: u16,
    pub event_id: Uuid,
    pub origin: Uuid,
    pub boot_id: Uuid,
    pub seq: u64,
    pub lamport: u64,
    pub kind: EventKind,
    pub content_hash: [u8; 32],
    pub size: u64,
}

impl ClipboardEvent {
    pub fn new(
        origin: Uuid,
        boot_id: Uuid,
        seq: u64,
        lamport: u64,
        kind: EventKind,
        content_hash: [u8; 32],
        size: u64,
    ) -> Self {
        Self {
            protocol_version: PROTOCOL_VERSION,
            event_id: Uuid::new_v4(),
            origin,
            boot_id,
            seq,
            lamport,
            kind,
            content_hash,
            size,
        }
    }
}

pub fn encode_event(event: &ClipboardEvent, max_len: usize) -> Result<Vec<u8>> {
    let body = serde_json::to_vec(event)
        .map_err(|e| CoreError::Protocol(format!("encode failed: {e}")))?;
    if body.len() > max_len {
        return Err(CoreError::Protocol(format!(
            "meta too large: {} > {max_len}",
            body.len()
        )));
    }
    let mut out = Vec::with_capacity(4 + body.len());
    out.extend_from_slice(&(body.len() as u32).to_be_bytes());
    out.extend_from_slice(&body);
    Ok(out)
}

pub fn decode_event(buf: &[u8], max_len: usize) -> Result<ClipboardEvent> {
    if buf.len() < 4 {
        return Err(CoreError::Protocol("meta header truncated".into()));
    }
    let len = u32::from_be_bytes([buf[0], buf[1], buf[2], buf[3]]) as usize;
    if len > max_len {
        return Err(CoreError::Protocol(format!(
            "meta too large: {len} > {max_len}"
        )));
    }
    if buf.len() != 4 + len {
        return Err(CoreError::Protocol(format!(
            "meta length mismatch: got {}, expected {}",
            buf.len(),
            4 + len
        )));
    }
    serde_json::from_slice(&buf[4..])
        .map_err(|e| CoreError::Protocol(format!("decode failed: {e}")))
}
```

```rust
    pub fn new() -> Self {
        Self::default()
    }

    pub fn tick_local(&mut self) -> u64 {
        self.value += 1;
        self.value
    }

    pub fn observe(&mut self, remote: u64) -> u64 {
        self.value = self.value.max(remote) + 1;
        self.value
    }

    pub fn value(&self) -> u64 {
        self.value
    }
}
```

```rust
use std::collections::{HashSet, VecDeque};

use uuid::Uuid;

use crate::protocol::ClipboardEvent;

#[derive(Debug)]
pub struct DedupWindow {
    boot_id: Option<Uuid>,
    last_seq: u64,
    seen: HashSet<Uuid>,
    order: VecDeque<Uuid>,
    capacity: usize,
}

impl DedupWindow {
    pub fn new(capacity: usize) -> Self {
        Self {
            boot_id: None,
            last_seq: 0,
            seen: HashSet::new(),
            order: VecDeque::new(),
            capacity: capacity.max(1),
        }
    }

    pub fn accept(&mut self, event: &ClipboardEvent) -> bool {
        if self.boot_id != Some(event.boot_id) {
            self.boot_id = Some(event.boot_id);
            self.last_seq = 0;
            self.seen.clear();
            self.order.clear();
        }

        if self.seen.contains(&event.event_id) {
            return false;
        }

        if event.seq + self.capacity as u64 <= self.last_seq {
            return false;
        }

        self.seen.insert(event.event_id);
        self.order.push_back(event.event_id);
        self.last_seq = self.last_seq.max(event.seq);

        while self.order.len() > self.capacity {
            if let Some(old) = self.order.pop_front() {
                self.seen.remove(&old);
            }
        }

        true
    }
}
```
